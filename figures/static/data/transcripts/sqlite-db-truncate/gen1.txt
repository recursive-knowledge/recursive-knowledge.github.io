import struct
import json
import re

# Read the truncated database
with open('trunc.db', 'rb') as f:
    data = f.read()

print("Analyzing SQLite structure and varint encoding...")

# Find all testword occurrences
pattern = b'testword'
matches = list(re.finditer(pattern, data))
records = []

for match in matches:
    start = match.start()
    word_bytes = data[start:start+10]
    word = word_bytes.decode('ascii', errors='ignore')
    
    # After the word, look for the value
    # In SQLite, values are often encoded as varints
    # Byte immediately after word can indicate type/length
    offset = start + 10
    
    if offset < len(data):
        first_byte = data[offset]
        
        # If first byte is small (< 128), it might be the value directly
        # Otherwise it could be part of varint encoding
        
        # Common pattern: single byte value
        if first_byte < 128:
            value = first_byte
        else:
            # Try to decode varint (SQLite varint encoding)
            val = 0
            shift = 0
            temp_offset = offset
            while temp_offset < len(data) and shift < 32:
                byte = data[temp_offset]
                val |= (byte & 0x7f) << shift
                temp_offset += 1
                if byte < 128:
                    break
                shift += 7
            value = val
        
        records.append({"word": word, "value": int(value)})
        print(f"{word}: value={value} (byte at offset: 0x{first_byte:02x})")

# Sort by word to ensure consistent order
records.sort(key=lambda x: x['word'])

with open('recover.json', 'w') as f:
    json.dump(records, f)

print(f"\nWrote {len(records)} records to recover.json")
