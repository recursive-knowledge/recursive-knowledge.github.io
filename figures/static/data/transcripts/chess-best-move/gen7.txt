def get_square_brightness(file, rank):
    """Get average brightness of a square"""
    x = file * square_size
    y = (7 - rank) * square_size
    
    total_brightness = 0
    count = 0
    for i in range(x, x + square_size):
        for j in range(y, y + square_size):
            if i < width and j < height:
                p = pixels[i, j]
                # Convert RGB to brightness
                if isinstance(p, tuple):
                    brightness = (p[0] * 299 + p[1] * 587 + p[2] * 114) // 1000
                else:
                    brightness = p
                total_brightness += brightness
                count += 1

    """Detect if there's a piece at the square and its type"""
    x = file * square_size

    # Check center region for piece detection
    center_x = x + square_size // 2
    center_y = y + square_size // 2

    colors = []

def analyze_square(file, rank):
    """Analyze a square to detect if there's a piece and what color it is"""
    x = file * square_size
    y = (7 - rank) * square_size
    
    # Collect all pixels in the square
    r_vals = []
    g_vals = []
    b_vals = []
    
    for i in range(x, x + square_size):
        for j in range(y, y + square_size):
            if i < width and j < height:
                p = pixels[i, j]
                if isinstance(p, tuple) and len(p) >= 3:
                    r_vals.append(p[0])
                    g_vals.append(p[1])
                    b_vals.append(p[2])
    
    if not r_vals:
        return None, None
    
    # Calculate statistics
    avg_r = sum(r_vals) // len(r_vals)
    avg_g = sum(g_vals) // len(g_vals)
    avg_b = sum(b_vals) // len(b_vals)

    var_r = sum((x - avg_r) ** 2 for x in r_vals) // len(r_vals)
    var_g = sum((x - avg_g) ** 2 for x in g_vals) // len(g_vals)

    # Brightness
    brightness = (avg_r * 299 + avg_g * 587 + avg_b * 114) // 1000
    
    return (avg_r, avg_g, avg_b, brightness, total_variance), None

    file = square_name[0]
