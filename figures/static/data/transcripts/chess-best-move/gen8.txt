# Analyze the chess board image more thoroughly to detect piece positions and colors by examining pixel regions

from PIL import Image
import numpy as np
import chess

img = Image.open('/app/chess_board.png')
img_array = np.array(img)

board_size = 8
square_size = 80

# Analyze the entire image more carefully, including larger areas
# to detect piece shapes and colors

def analyze_square(img_array, file, rank):
    """Analyze a square to detect piece and color"""
    x_start = file * square_size
    y_start = (board_size - 1 - rank) * square_size
    x_end = x_start + square_size
    y_end = y_start + square_size
    
    square_region = img_array[y_start:y_end, x_start:x_end]
    
    # Sample multiple points in the square
    colors = []
    for i in range(10, 70, 20):
        for j in range(10, 70, 20):
            colors.append(tuple(square_region[i, j]))
    
    return square_region, colors

# Build a visual representation first
print("Raw analysis of each square:")
board_map = {}
for rank in range(board_size):
    for file in range(board_size):
        square_region, colors = analyze_square(img_array, file, rank)
        square_name = chr(97 + file) + str(rank + 1)
        
        # Count dark vs light pixels
        dark_count = 0
        light_count = 0
        
        for i in range(square_size):
            for j in range(square_size):
                r, g, b = square_region[i, j]
                brightness = (r + g + b) // 3
                if brightness < 100:
                    dark_count += 1
                else:
                    light_count += 1
        
        board_map[square_name] = {
            'dark': dark_count,
            'light': light_count,
            'sample_colors': colors[:3]
        }

# Identify pieces
print("\nSquares with significant dark regions (likely pieces):")
for square, info in sorted(board_map.items()):
    if info['dark'] > 500:  # threshold for occupied square
        print(f"{square}: dark_pixels={info['dark']}, light_pixels={info['light']}")

    width = cols[-1] - cols[0] + 1

    top_filled = np.sum(piece_region[0, :])

    # Calculate aspect ratio and fill density
    aspect_ratio = width / height if height > 0 else 1
    fill_density = area / (height * width) if height * width > 0 else 0
