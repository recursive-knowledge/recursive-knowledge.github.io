import fasttext
import pandas as pd
import os

# Load the training data
train_df = pd.read_parquet('/app/data/train-00000-of-00001.parquet')
test_df = pd.read_parquet('/app/data/test-00000-of-00001.parquet')

# Convert to fastText format: __label__<label> <text>
train_file = '/tmp/train.txt'
test_file = '/tmp/test.txt'

print("Preparing training data...")
with open(train_file, 'w') as f:
    for _, row in train_df.iterrows():
        label = row['label']
        text = row['text'].replace('\n', ' ').strip()
        f.write(f"__label__{label} {text}\n")

print("Preparing test data...")
with open(test_file, 'w') as f:
    for _, row in test_df.iterrows():
        label = row['label']
        text = row['text'].replace('\n', ' ').strip()
        f.write(f"__label__{label} {text}\n")

print(f"Training file size: {os.path.getsize(train_file) / (1024*1024):.2f} MB")
print(f"Test file size: {os.path.getsize(test_file) / (1024*1024):.2f} MB")

# Train fastText model with focus on keeping size small and accuracy high
print("\nTraining fastText model...")
model = fasttext.train_supervised(
    input=train_file,
    epoch=25,
    lr=1.0,
    wordNgrams=2,
    minn=3,
    maxn=6,
    dim=100
)

# Evaluate on test set
N, precision, recall = model.test(test_file)
accuracy = precision
print(f"\nTest accuracy: {accuracy:.4f}")
print(f"Test samples: {N}")

# Save the model
model_path = '/app/model.bin'
model.save_model(model_path)
model_size_mb = os.path.getsize(model_path) / (1024*1024)
print(f"\nModel saved to {model_path}")
print(f"Model size: {model_size_mb:.2f} MB")

if model_size_mb < 150 and accuracy >= 0.62:
    print("\n✓ Model meets requirements!")
else:
    print(f"\n✗ Model does NOT meet requirements:")
    print(f"  Size: {model_size_mb:.2f} MB (limit: 150 MB) - {'PASS' if model_size_mb < 150 else 'FAIL'}")
    print(f"  Accuracy: {accuracy:.4f} (target: >= 0.62) - {'PASS' if accuracy >= 0.62 else 'FAIL'}")

import pandas as pd

train_file = '/tmp/train_sampled.txt'
test_file = '/tmp/test.txt'

# Process training data in batches with sampling
with open(train_file, 'w') as f:
    for batch_idx, batch in enumerate(parquet_file.iter_batches(batch_size=10000)):
        df = batch.to_pandas()
        # Keep every 5th row (20% sampling)
        for i, (_, row) in enumerate(df.iterrows()):

                label = row['label']
                text = row['text'].replace('\n', ' ').strip()
