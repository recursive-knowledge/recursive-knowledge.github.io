#!/usr/bin/env python3
import os
import pytesseract
from PIL import Image
import re

# Directory containing frames
frames_dir = '/app/frames'
output_file = '/app/solution.txt'

# Collect all text from frames
all_text = []
frame_files = sorted([f for f in os.listdir(frames_dir) if f.endswith('.png')])

print(f"Processing {len(frame_files)} frames...")

for i, frame_file in enumerate(frame_files):
    frame_path = os.path.join(frames_dir, frame_file)
    try:
        img = Image.open(frame_path)
        text = pytesseract.image_to_string(img)
        if text.strip():
            all_text.append((i+1, frame_file, text))
            if (i+1) % 20 == 0:
                print(f"  Processed frame {i+1}")
    except Exception as e:
        print(f"Error processing {frame_file}: {e}")

print(f"\nExtracted text from {len(all_text)} frames")

# Now parse moves from the extracted text
# Zork commands are typically short: n, s, e, w, get <item>, drop <item>, etc.
moves = []

# Combine all text
combined_text = '\n'.join([text for _, _, text in all_text])

# Look for patterns in the text that look like commands
# These might appear as prompts or in input areas
lines = combined_text.split('\n')

for line in lines:
    line = line.strip()
    if not line:
        continue
    
    # Common Zork commands/patterns
    # Match simple directional commands
    if re.match(r'^[news]$', line, re.IGNORECASE):
        moves.append(line.lower())
    # Match "get <item>", "drop <item>", etc.
    elif re.match(r'^(get|drop|take|examine|look|inventory|help|quit|load|save)\b', line, re.IGNORECASE):
        moves.append(line.lower())
    # Match directions with modifiers
    elif re.match(r'^(north|south|east|west|up|down|in|out)\b', line, re.IGNORECASE):
        moves.append(line.lower())

print(f"\nFound {len(moves)} potential moves")
print("First 20 moves:", moves[:20])

# Write to solution file
with open(output_file, 'w') as f:
    for move in moves:
        f.write(move + '\n')

print(f"\nWrote {len(moves)} moves to {output_file}")

#!/usr/bin/env python3
import cv2
import pytesseract
import os
import re
from collections import defaultdict

# Use OpenCV to process video more efficiently
video_path = '/app/zork_video.mp4'
output_file = '/app/solution.txt'

# Open video
cap = cv2.VideoCapture(video_path)
fps = cap.get(cv2.CAP_PROP_FPS)
frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

print(f"Video: {frame_count} frames at {fps} fps")

# Sample every N frames to speed up processing
sample_rate = 10
frame_idx = 0
all_moves = []
move_set = set()

while True:
    ret, frame = cap.read()
    if not ret:
        break
    
    # Only process sampled frames
    if frame_idx % sample_rate == 0:
        # Crop to bottom portion where commands appear (typical Zork UI)
        h, w = frame.shape[:2]
        # Focus on bottom ~25% where text input typically appears
        bottom_crop = frame[int(h*0.75):h, :]
        
        # Convert to grayscale and apply threshold for better OCR
        gray = cv2.cvtColor(bottom_crop, cv2.COLOR_BGR2GRAY)
        _, thresh = cv2.threshold(gray, 150, 255, cv2.THRESH_BINARY)
        
        # OCR on the cropped area
        try:
            text = pytesseract.image_to_string(thresh)
            # Extract lines and look for command patterns
            for line in text.split('\n'):
                line = line.strip()
                if not line or len(line) > 50:  # Skip very long lines (noise)
                    continue
                
                # Match Zork command patterns
                # Single direction
                if re.match(r'^[newsudio]$', line, re.IGNORECASE):
                    cmd = line.lower()
                    if cmd not in move_set:
                        all_moves.append(cmd)
                        move_set.add(cmd)
                # Multi-word commands like "get bag", "go west"
                elif re.match(r'^(get|drop|take|examine|look|go|north|south|east|west|up|down|in|out|open|close|put)\s+\w+', line, re.IGNORECASE):
                    cmd = line.lower()
                    if cmd not in move_set:
                        all_moves.append(cmd)
                        move_set.add(cmd)
        except Exception as e:
            pass
    
    frame_idx += 1
    if frame_idx % 50 == 0:
        print(f"  Processed {frame_idx}/{frame_count} frames, found {len(all_moves)} moves so far")

cap.release()

print(f"\nFound {len(all_moves)} unique moves")
print("Sample:", all_moves[:10] if all_moves else "None")

# Write to solution
with open(output_file, 'w') as f:
    for move in all_moves:
        f.write(move + '\n')

print(f"Wrote to {output_file}")

    text_lower = text.lower()
    
    # Single letters for directions
    if re.match(r'^[newsudio]$', text_lower):
        return True
    
    # Direction words

        r'^quit$',
    ]
    
    for pattern in patterns:

            return True
    
    return False
