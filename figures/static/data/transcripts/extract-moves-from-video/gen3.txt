#!/usr/bin/env python3
import cv2
import pytesseract
import os
import re

# Extract frames from video every half second
video_path = '/app/zork_video.mp4'
output_dir = '/app/frames'

os.makedirs(output_dir, exist_ok=True)

cap = cv2.VideoCapture(video_path)
fps = cap.get(cv2.CAP_PROP_FPS)
frame_interval = int(fps * 0.5)  # Every 0.5 seconds

frame_count = 0
extracted_count = 0

all_text = []

while True:
    ret, frame = cap.read()
    if not ret:
        break
    
    if frame_count % frame_interval == 0:
        frame_path = os.path.join(output_dir, f'frame_{extracted_count:04d}.png')
        cv2.imwrite(frame_path, frame)
        
        # OCR the frame
        try:
            text = pytesseract.image_to_string(frame)
            if text.strip():
                all_text.append(text)
        except Exception as e:
            print(f"Error OCRing frame {extracted_count}: {e}")
        
        extracted_count += 1
    
    frame_count += 1

cap.release()

# Combine all extracted text
full_text = '\n'.join(all_text)

# Save raw text for inspection
with open('/app/raw_text.txt', 'w') as f:
    f.write(full_text)

print(f"Extracted {extracted_count} frames and performed OCR")
print(f"Full text saved to /app/raw_text.txt")
EOF
