from sentence_transformers import SentenceTransformer
import numpy as np

# Load the model at the specified revision
model = SentenceTransformer('BAAI/bge-small-zh-v1.5', revision='7999e1d3359715c523056ef9478215996d62a620')

# Read all documents from /app/data.txt
with open('/app/data.txt', 'r', encoding='utf-8') as f:
    documents = [line.strip() for line in f.readlines() if line.strip()]

print(f"Total documents: {len(documents)}")

# Encode query
query = "terminal-bench"
query_embedding = model.encode(query, normalize_embeddings=True)

# Encode all documents
doc_embeddings = model.encode(documents, normalize_embeddings=True)

# Calculate cosine similarities (using normalized embeddings, dot product = cosine similarity)
similarities = np.dot(doc_embeddings, query_embedding)

# Get indices sorted by similarity (descending)
sorted_indices = np.argsort(similarities)[::-1]

# Get top similarities and documents
print("\nTop 10 documents by cosine similarity:")
for rank, idx in enumerate(sorted_indices[:10], 1):
    print(f"{rank}. Similarity: {similarities[idx]:.6f}")
    print(f"   Document: {documents[idx][:100]}")

# The 5th highest similarity document (index 4 in sorted list)
fifth_idx = sorted_indices[4]
fifth_doc = documents[fifth_idx]

print(f"\n5th highest cosine similarity document:")
print(f"Index in original: {fifth_idx}")
print(f"Similarity score: {similarities[fifth_idx]:.6f}")
print(f"Document: {fifth_doc}")

# Write result to /app/result.txt
with open('/app/result.txt', 'w', encoding='utf-8') as f:
    f.write(fifth_doc)

print("\nResult written to /app/result.txt")
