#include<stdio.h>

typedef struct{float w[H][H],b[H];}QK;
typedef struct{float w[H][V],b[V];}EM;

float*buf[4];

void read_ckpt(char*f){
FILE*p=fopen(f,"rb");
fseek(p,0,SEEK_END);
long sz=ftell(p);
fseek(p,0,SEEK_SET);
float*d=malloc(sz);
fread(d,1,sz,p);
fclose(p);
buf[0]=d;
}

void read_bpe(char*f,char**vocab){
FILE*p=fopen(f,"r");
char ln[256];
int i=0;
while(fgets(ln,256,p)&&i<V){
ln[strcspn(ln,"\n")]=0;
vocab[i]=strdup(ln);
i++;
}
fclose(p);
}

int softmax_argmax(float*logits,int n){
float mx=-INFINITY;
int idx=0;
for(int i=0;i<n;i++)if(logits[i]>mx){mx=logits[i];idx=i;}
return idx;
}

int main(int argc,char**argv){
if(argc<4)return 1;
char*ckpt=argv[1],*bpe_f=argv[2],*prompt=argv[3];
char**vocab=malloc(V*sizeof(char*));
read_ckpt(ckpt);
read_bpe(bpe_f,vocab);
float*logits=malloc(V*sizeof(float));
memset(logits,0,V*sizeof(float));
for(int t=0;t<20;t++){
int tid=softmax_argmax(logits,V);
printf("%s",vocab[tid]);
}
printf("\n");
return 0;
}
