Model model;

void load_vocab(const char *path) {
    FILE *f = fopen(path, "r");
    char line[256];
    vocab_len = 0;
    while (fgets(line, sizeof(line), f) && vocab_len < 50001) {
        line[strcspn(line, "\n")] = 0;
        if (line[0] == '#') continue;

void load_checkpoint(const char *path) {
    FILE *f = fopen(path, "rb");
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    fseek(f, 0, SEEK_SET);
    
    float *data = malloc(size);
    fread(data, 1, size, f);
    fclose(f);
    
    model.wte = data;
    int offset = VOCAB_SIZE * EMBED_DIM;
    model.wpe = data + offset;
    
    free(data);
}

int argmax(float *logits, int n) {
    int idx = 0;
    float max_val = logits[0];
    for (int i = 1; i < n; i++) {
        if (logits[i] > max_val) {
            max_val = logits[i];
            idx = i;
        }
    }
    return idx;
}

void forward(TokenSeq *seq, float *output) {
    memset(output, 0, VOCAB_SIZE * sizeof(float));
    if (seq->len > 0) {
        int last_token = seq->tokens[seq->len - 1];
        if (last_token >= 0 && last_token < VOCAB_SIZE) {
            float *embed = model.wte + last_token * EMBED_DIM;
            for (int i = 0; i < VOCAB_SIZE && i < EMBED_DIM; i++) {
                output[i] = embed[i];
            }
        }
    }
}

        }
    }
