} Matrix;

    Matrix wte, wpe;
    Matrix ln1w, ln1b, attn_w, attn_b, mlp_fc_w, mlp_fc_b, mlp_proj_w, mlp_proj_b;
    Matrix ln2w, ln2b;
    Matrix ln_f_w, ln_f_b;
} Model;

int read_ckpt(const char *path, Model *m) {
    FILE *f = fopen(path, "rb");
    if (!f) return 0;
    
    // Read word embeddings (50257 x 768)
    m->wte.rows = VOCAB_SIZE; m->wte.cols = EMBED_DIM;
    m->wte.data = malloc(VOCAB_SIZE * EMBED_DIM * sizeof(float));
    fread(m->wte.data, sizeof(float), VOCAB_SIZE * EMBED_DIM, f);
    
    // Read position embeddings (512 x 768)
    m->wpe.rows = MAX_SEQ; m->wpe.cols = EMBED_DIM;
    m->wpe.data = malloc(MAX_SEQ * EMBED_DIM * sizeof(float));
    fread(m->wpe.data, sizeof(float), MAX_SEQ * EMBED_DIM, f);
    
    fclose(f);
    return 1;
}

int read_vocab(const char *path, char **vocab) {
    FILE *f = fopen(path, "r");
    if (!f) return 0;
    
    char line[256];
    int idx = 0;
    fgets(line, sizeof(line), f); // skip version line
    
    while (fgets(line, sizeof(line), f) && idx < MAX_VOCAB) {
        line[strcspn(line, "\n")] = 0;
        vocab[idx] = malloc(strlen(line) + 1);
        strcpy(vocab[idx], line);
        idx++;
    }
    fclose(f);
    return idx;
}

float softmax_sample(float *logits, int size) {
    float max_logit = logits[0];
    for (int i = 1; i < size; i++)
        if (logits[i] > max_logit) max_logit = logits[i];
    
    float sum = 0;
    for (int i = 0; i < size; i++) {
        logits[i] = exp(logits[i] - max_logit);
        sum += logits[i];
    }
    
    int argmax = 0;
    float max_prob = logits[0];
    for (int i = 1; i < size; i++)
        if (logits[i] > max_prob) {
            max_prob = logits[i];
            argmax = i;
        }
    return argmax;
}

int main(int argc, char *argv[]) {
    if (argc < 4) {
        fprintf(stderr, "Usage: %s <ckpt> <vocab> <text>\n", argv[0]);
        return 1;
    }
    
    Model m = {0};
    char *vocab[MAX_VOCAB];
    
    if (!read_ckpt(argv[1], &m)) {
        fprintf(stderr, "Failed to read checkpoint\n");
        return 1;
    }
    
    int vocab_size = read_vocab(argv[2], vocab);
    if (vocab_size <= 0) {
        fprintf(stderr, "Failed to read vocab\n");
        return 1;
    }
    
    printf("%s", argv[3]);

    float logits[MAX_VOCAB];
    for (int i = 0; i < 20; i++) {
        for (int j = 0; j < vocab_size; j++) {

        if (next_token < vocab_size && vocab[next_token]) {
            printf("%s", vocab[next_token]);
            fflush(stdout);
        }
    }
    printf("\n");
    
    return 0;
}

#define L 12
#define S 1024

float wte[V*E], wpe[S*E];
float w[L*E*4*E], b[L*E];
char *vocab[V];
int vn;

int readckpt(char *path){
 FILE *f=fopen(path,"rb");
 if(!f)return 0;
 int ok=1;
 ok&=fread(wte,4,V*E,f)==V*E;
 ok&=fread(wpe,4,S*E,f)==S*E;
 for(int i=0;i<L;i++){
  ok&=fread(w+i*E*4*E,4,E*4*E,f)==E*4*E;
  ok&=fread(b+i*E,4,E,f)==E;
 }
 fclose(f);
 return ok;
}

int readvocab(char *path){
 FILE *f=fopen(path,"r");
 if(!f)return 0;
 char line[512];
 if(!fgets(line,512,f))return 0;
 for(vn=0;vn<V&&fgets(line,512,f);vn++){
  line[strcspn(line,"\n")]=0;
  vocab[vn]=malloc(strlen(line)+1);
  strcpy(vocab[vn],line);
 }
 fclose(f);
 return vn;
}

int tok(char*s,int*t){
 int n=0;
 while(*s&&n<S){
  for(int i=0;i<vn;i++){
   if(!strncmp(s,vocab[i],strlen(vocab[i]))){
    s+=strlen(vocab[i]);
    t[n++]=i;
    goto next;
   }
  }
  s++;
  next:;
 }
 return n;
}

 if(argc<4)return 1;

 if(!readvocab(argv[2]))return 1;
 printf("%s",argv[3]);
 fflush(stdout);
 int t[S];

  for(int d=0;d<E;d++){
   h[d]=wte[t[n-1]*E+d]+wpe[(n-1)*E+d];
   for(int j=0;j<L;j++)h[d]+=w[j*E*4*E+d*E];

  }

 }
 printf("\n");
 return 0;
}
