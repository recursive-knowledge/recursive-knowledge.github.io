    """Classify a pixel as empty or containing a piece"""

from PIL import Image
import numpy as np

# Load the chess board image
img = Image.open('chess_board.png')
img_array = np.array(img, dtype=np.uint8)

square_size = 80

# For each square, sample multiple pixels to detect if there's a piece
print("Analyzing each square for piece detection:")
print("Using variance of pixel colors in each square to detect pieces")
print()

board_map = []
for row in range(8):
    row_data = []
    for col in range(8):
        # Extract the square region
        y_start = row * square_size
        y_end = (row + 1) * square_size
        x_start = col * square_size
        x_end = (col + 1) * square_size
        
        square_region = img_array[y_start:y_end, x_start:x_end, :]
        
        # Calculate variance of colors in this square
        variance = np.var(square_region)
        
        # Count pixels that are very dark (0,0,0) or very different
        dark_pixels = np.sum((square_region[:,:,0] < 50) & (square_region[:,:,1] < 50) & (square_region[:,:,2] < 50))
        
        # Sample center pixel
        center_y = square_size // 2
        center_x = square_size // 2
        center_pixel = tuple(square_region[center_y, center_x])
        
        row_data.append({
            'col': col,
            'row': row,
            'variance': variance,
            'dark_pixels': dark_pixels,
            'center_color': center_pixel
        })
        
        status = "PIECE" if variance > 500 or dark_pixels > 100 else "empty"
        print(f"({col},{row}): var={variance:6.0f}, dark={dark_pixels:3d}, center={center_pixel} -> {status}")
    
    board_map.append(row_data)
