# The image is 640x640, representing an 8x8 board
square_size = 80

# Create a detailed analysis by examining larger regions of each square
# to identify pieces more reliably
def analyze_square(img, row, col, square_size):
    """Analyze a single square to determine what piece it contains"""
    x_start = col * square_size
    y_start = row * square_size
    x_end = x_start + square_size
    y_end = y_start + square_size
    
    # Crop the square
    square_img = img.crop((x_start, y_start, x_end, y_end))
    
    # Sample multiple points to get a better reading
    pixels = []
    for dy in [20, 40, 60]:
        for dx in [20, 40, 60]:
            pixels.append(square_img.getpixel((dx, dy)))
    
    return pixels

# Analyze all squares

for row in range(8):
    board_map.append([])

        counter = Counter(pixels)

# Strategy: look for checks, checkmates, and forcing moves
checkmates = []
checks = []
captures = []

for move in legals:
    test = board.copy()
    cap = test.piece_at(move.to_square)
    test.push(move)
    
    if test.is_checkmate():

    elif test.is_check():

        val = {1:1, 2:3, 3:3, 4:5, 5:9}.get(cap.piece_type, 0)
