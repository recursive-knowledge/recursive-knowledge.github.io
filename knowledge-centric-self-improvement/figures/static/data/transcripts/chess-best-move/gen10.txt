# Based on the pixel analysis, I need to identify:
# - Empty squares: light (240,217,181) or dark (181,136,99)
# - Black pieces: very dark (0,0,0) or very dark brown
# - White pieces: very light (255,255,255) or light colors

board = [['.' for _ in range(8)] for _ in range(8)]

for row in range(8):
    for col in range(8):
        x = col * 80 + 40
        y = row * 80 + 40
        r, g, b = pixels[x, y]
        
        # Classify based on color
        if (r, g, b) == (0, 0, 0):
            # Pure black - black piece
            board[row][col] = 'bp'  # black piece (need to determine type)
        elif r > 240 and g > 240 and b > 240:
            # Pure white or near-white - white piece
            board[row][col] = 'wp'  # white piece
        elif r > 200 and g > 180 and b > 150:
            # Light square, likely empty
            board[row][col] = ' '
        elif r < 100 and g < 100 and b < 100 and (r, g, b) != (0, 0, 0):
            # Dark piece
            board[row][col] = 'bp'

            # Ambiguous - might be piece on dark square
            board[row][col] = '?'

            board[row][col] = ' '
