#!/bin/bash
set -e

# Kill any existing qemu processes
for pid in $(grep -l qemu /proc/*/cmdline 2>/dev/null | xargs -I{} dirname {} | xargs basename); do
  [ "$pid" != "self" ] && [ "$pid" != "thread-self" ] && kill -9 "$pid" 2>/dev/null || true
done

# Wait for port to become free
echo "Waiting for port 6665 to be released..."
for i in {1..30}; do
  if ! grep -q '1A09' /proc/net/tcp 2>/dev/null; then
    echo "Port 6665 is free"
    break
  fi
  echo "Waiting... ($i/30)"
  sleep 1
done

# Ensure disk image is fresh
rm -f /app/alpine-disk.qcow2
cp /app/alpine-disk.qcow2.backup /app/alpine-disk.qcow2

# Start QEMU with serial over TCP on port 6665
# Use -chardev and -serial to bind serial console to TCP port
/usr/bin/qemu-system-x86_64 \
  -cdrom /app/alpine.iso \
  -hda /app/alpine-disk.qcow2 \
  -m 512 \
  -nographic \
  -boot d \
  -chardev socket,id=char0,host=127.0.0.1,port=6665,server,nowait \
  -serial chardev:char0 \
  -monitor none \
  > /tmp/qemu.log 2>&1 &

QEMU_PID=$!
echo $QEMU_PID > /tmp/qemu.pid
echo "QEMU started with PID $QEMU_PID"

# Wait for Alpine to boot and show login prompt
echo "Waiting for Alpine to boot..."
for i in {1..120}; do
  if timeout 2 bash -c "echo > /dev/tcp/127.0.0.1/6665" 2>/dev/null; then
    echo "Port 6665 is responding"
    # Give it a bit more time to show the login prompt
    sleep 5
    timeout 2 telnet 127.0.0.1 6665 2>&1 | head -10 && break
  fi
  echo "Waiting for port to be ready ($i/120)..."
  sleep 1
done

echo "QEMU startup complete and ready for telnet connection"
EOF
chmod +x /tmp/start_qemu.sh
