#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define VOCAB_SIZE 50257
#define EMBED_DIM 768
#define MAX_SEQ 1024

typedef struct {
    float *wte;    // token embedding [VOCAB_SIZE, EMBED_DIM]
    float *wpe;    // position embedding [MAX_SEQ, EMBED_DIM]
    float *ln1w, *ln1b;  // layer norm weights/bias
    float *attn_w;       // attention weights
    float *attn_b;       // attention bias
    float *mlp_w;        // mlp weights
    float *mlp_b;        // mlp bias
    float *ln2w, *ln2b;
    float *lm_head;      // language model head
} Model;

void load_bpe(const char *path, char **tokens) {
    FILE *f = fopen(path, "r");
    if (!f) return;
    
    char line[256];
    int idx = 0;
    while (fgets(line, sizeof(line), f) && idx < VOCAB_SIZE) {
        int len = strlen(line);
        if (len > 0 && line[len-1] == '\n') line[len-1] = 0;
        if (line[0] == '#') continue;
        tokens[idx] = malloc(strlen(line) + 1);
        strcpy(tokens[idx], line);
        idx++;
    }
    fclose(f);
}

void load_ckpt(const char *path, Model *m) {
    FILE *f = fopen(path, "rb");
    if (!f) return;
    
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    fseek(f, 0, SEEK_SET);
    
    float *data = malloc(size);
    fread(data, 1, size, f);
    fclose(f);
    
    int pos = 0;
    m->wte = data + pos; pos += VOCAB_SIZE * EMBED_DIM;
    m->wpe = data + pos; pos += MAX_SEQ * EMBED_DIM;
    m->ln1w = data + pos; pos += 12 * EMBED_DIM;
    m->ln1b = data + pos; pos += 12 * EMBED_DIM;
    m->attn_w = data + pos; pos += 12 * EMBED_DIM * EMBED_DIM * 3;
    m->attn_b = data + pos; pos += 12 * EMBED_DIM * 3;
    m->mlp_w = data + pos; pos += 12 * EMBED_DIM * EMBED_DIM * 4 * 2;
    m->mlp_b = data + pos; pos += 12 * EMBED_DIM * (4 + 1);
    m->ln2w = data + pos; pos += 12 * EMBED_DIM;
    m->ln2b = data + pos; pos += 12 * EMBED_DIM;
    m->lm_head = data + pos;
}

int sample_token(float *logits) {
    int best = 0;
    float best_val = logits[0];
    for (int i = 1; i < VOCAB_SIZE; i++) {
        if (logits[i] > best_val) {
            best_val = logits[i];
            best = i;
        }
    }
    return best;
}

int main(int argc, char *argv[]) {
    if (argc < 4) return 1;
    
    const char *ckpt = argv[1];
    const char *bpe = argv[2];
    const char *prompt = argv[3];
    
    Model m = {0};
    load_ckpt(ckpt, &m);
    
    char **tokens = malloc(VOCAB_SIZE * sizeof(char*));
    load_bpe(bpe, tokens);
    
    float logits[VOCAB_SIZE];
    for (int i = 0; i < 20; i++) {
        memset(logits, 0, sizeof(logits));
        int next = sample_token(logits);
        if (tokens[next]) printf("%s", tokens[next]);
    }
    printf("\n");
    
    return 0;
}

float *forward(const char *prompt) {

    float *hidden = (float*)calloc(D, sizeof(float));
