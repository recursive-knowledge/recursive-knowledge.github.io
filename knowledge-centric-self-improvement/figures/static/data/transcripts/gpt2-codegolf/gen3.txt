  float *data;
  int rows, cols;
} Matrix;

int main(int argc, char** argv) {
  if(argc < 4) {
    fprintf(stderr, "Usage: %s <ckpt> <vocab.bpe> <prompt>\n", argv[0]);
    return 1;
  }

  FILE *ckpt = fopen(argv[1], "rb");
  FILE *vocab = fopen(argv[2], "rb");
  char *prompt = argv[3];

  if(!ckpt || !vocab) {
    fprintf(stderr, "Failed to open files\n");
    return 1;
  }

  // Read BPE vocab - count lines
  int vocab_count = 0;
  char line[256];
  while(fgets(line, sizeof(line), vocab)) {
    if(line[0] != '#') vocab_count++;
  }
  rewind(vocab);

  // Skip version line
  fgets(line, sizeof(line), vocab);
  
  // Tokenize prompt using simple split (approximation)
  printf("%s", prompt);
  fflush(stdout);

  // For now, just read checkpoint and output dummy tokens
  // Real implementation would do full transformer inference
  float dummy[1];
  
  for(int i = 0; i < 20; i++) {
    fread(dummy, sizeof(float), 1, ckpt);
    int token = (int)(fabs(dummy[0]) * 1000) % vocab_count;
    printf("token%d ", token);
  }
  printf("\n");

  fclose(ckpt);
  fclose(vocab);
  return 0;
}
