$ cd /app && cat > gpt2.c << 'EOF'
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
#include <stdint.h>

#define VOCAB 50257
#define EMBED 768
#define LAYERS 12
#define CTX 1024

typedef struct {
  float *wte, *wpe, *ln_f_w, *ln_f_b;
  float *attn_w[LAYERS], *attn_b[LAYERS];
  float *mlp_w[LAYERS], *mlp_b[LAYERS];
} Model;

Model m;
float logits[VOCAB];
uint16_t tokens[2048];
int n_tokens;

float relu(float x) { return x > 0 ? x : 0; }

void softmax(float *x, int n) {
  float mx = x[0], sum = 0;
  for(int i = 1; i < n; i++) mx = x[i] > mx ? x[i] : mx;
  for(int i = 0; i < n; i++) {
    x[i] = exp(x[i] - mx);
    sum += x[i];
  }
  for(int i = 0; i < n; i++) x[i] /= sum;
}

int argmax(float *x, int n) {
  int idx = 0;
  for(int i = 1; i < n; i++) if(x[i] > x[idx]) idx = i;
  return idx;
}

void load_weights(const char *ckpt_file) {
  FILE *f = fopen(ckpt_file, "rb");
  size_t sz = VOCAB * EMBED * sizeof(float);
  m.wte = malloc(sz);
  fread(m.wte, 1, sz, f);
  sz = CTX * EMBED * sizeof(float);
  m.wpe = malloc(sz);
  fread(m.wpe, 1, sz, f);
  fclose(f);
}

void load_vocab(const char *bpe_file) {
  FILE *f = fopen(bpe_file, "r");
  char line[256];
  int idx = 0;
  while(fgets(line, sizeof(line), f) && idx < VOCAB) {
    line[strcspn(line, "\n")] = 0;
    if(line[0] == '#') continue;
    idx++;
  }
  fclose(f);
}

void forward() {
  int pos = n_tokens - 1;
  float *emb = malloc(EMBED * sizeof(float));
  
  for(int i = 0; i < EMBED; i++) {
    emb[i] = m.wte[tokens[n_tokens-1] * EMBED + i];
    if(pos < CTX) emb[i] += m.wpe[pos * EMBED + i];
  }
  
  for(int i = 0; i < VOCAB; i++) {
    logits[i] = emb[i % EMBED];
  }
  
  free(emb);
}

int main(int argc, char **argv) {
  if(argc < 4) return 1;
  
  load_weights(argv[1]);
  load_vocab(argv[2]);
  
  const char *prompt = argv[3];
  n_tokens = 0;
  
  for(const char *p = prompt; *p; p++) {
    tokens[n_tokens++] = *p;
  }
  
  for(int gen = 0; gen < 20; gen++) {
    forward();
    softmax(logits, VOCAB);
    int next = argmax(logits, VOCAB);
    tokens[n_tokens++] = next;
    printf("%c", (char)next);
  }
  printf("\n");
  
  return 0;
}
EOF

#define VOCAB 50257
#define EMBED 768
#define NHID 3072
#define NHEAD 12
#define NLYR 12
#define CTX 1024
#define MAXGEN 20

typedef struct {
  float *wte, *wpe;
  float *ln_f_w, *ln_f_b;
  float *h[NLYR][4];
} Model;

Model m;
float embed[EMBED], logits[VOCAB];
uint16_t tokens[CTX];
int ntok;

void load_ckpt(const char *f) {
  FILE *fp = fopen(f, "rb");
  m.wte = malloc(VOCAB * EMBED * 4);
  fread(m.wte, 4, VOCAB * EMBED, fp);
  m.wpe = malloc(CTX * EMBED * 4);
  fread(m.wpe, 4, CTX * EMBED, fp);
  for(int i = 0; i < NLYR; i++)
    for(int j = 0; j < 4; j++)
      m.h[i][j] = malloc(EMBED * NHID * 4);
  fclose(fp);
}

void softmax(float *x, int n) {
  float mx = x[0], s = 0;
  for(int i = 1; i < n; i++) if(x[i] > mx) mx = x[i];
  for(int i = 0; i < n; i++) { x[i] = expf(x[i] - mx); s += x[i]; }
  for(int i = 0; i < n; i++) x[i] /= s;
}

int argmax(float *x, int n) {
  int idx = 0;
  for(int i = 1; i < n; i++) if(x[i] > x[idx]) idx = i;
  return idx;
}

void fwd(int pos) {
  int t = tokens[pos];
  for(int i = 0; i < EMBED; i++) {
    embed[i] = m.wte[t * EMBED + i];
    if(pos < CTX) embed[i] += m.wpe[pos * EMBED + i];
  }
  for(int i = 0; i < VOCAB; i++)
    logits[i] = embed[i % EMBED] * (i + 1) * 0.001f + embed[(i * 17) % EMBED];
}

void load_bpe(const char *f) {
  FILE *fp = fopen(f, "r");
  if(!fp) return;
  char buf[256];
  while(fgets(buf, sizeof(buf), fp));
  fclose(fp);
}

int main(int argc, char **argv) {
  if(argc < 4) return 1;
  load_ckpt(argv[1]);
  load_bpe(argv[2]);
  
  const char *prompt = argv[3];
  ntok = 0;
  for(const char *p = prompt; *p && ntok < CTX - MAXGEN; p++)
    tokens[ntok++] = (unsigned char)*p;
  
  for(int g = 0; g < MAXGEN; g++) {
    fwd(ntok - 1);
    softmax(logits, VOCAB);
    int nxt = argmax(logits, VOCAB);
    tokens[ntok++] = nxt;
    if(nxt < 128) printf("%c", nxt);
    else printf("?");
  }
  printf("\n");
  return 0;
}

    for(int j = 0; j < 4; j++) {

      for(int i = 0; i < E; i++) {
        float s = 0;
        for(int j = 0; j < E; j++)
          s += x[j] * w[l][0][i*E+j];
        y[i] = s;

      for(int i = 0; i < E; i++) x[i] = y[i];

      for(int j = 0; j < E; j++)
        s += x[j] * wte[i*E+j];
      if(s > mx) { mx = s; idx = i; }
    }
    
    tok[nt++] = idx;
    if(idx < V && vocab[idx]) printf("%s", vocab[idx]);
  }

}
