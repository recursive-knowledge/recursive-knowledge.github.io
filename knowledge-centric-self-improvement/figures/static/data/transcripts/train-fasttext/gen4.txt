model_path = '/app/model.bin'

print('Training fasttext model...')
# Using supervised learning with optimization for accuracy
# lr: learning rate, epoch: number of epochs, dim: dimension of word vectors
# minn, maxn: min/max length of character n-grams
model = fasttext.train_supervised(
    input=train_file,
    epoch=25,
    lr=1.0,
    wordNgrams=2,
    dim=100,
    loss='softmax'
)

print(f'Model trained. Saving to {model_path}...')
model.save_model(model_path)

print(f'Model saved. Checking size...')
model_size = os.path.getsize(model_path) / (1024 * 1024)
print(f'Model size: {model_size:.2f} MB')

# Evaluate on test set
print('\nEvaluating on test set...')
results = model.test(test_file)
print(f'Test set results: {results}')
accuracy = results[1]
print(f'Accuracy: {accuracy:.4f}')

# Check if meets constraints
if model_size < 150 and accuracy >= 0.62:
    print('\n✓ Model meets constraints: size < 150MB and accuracy >= 0.62')
else:
    print(f'\n✗ Model does NOT meet constraints:')
    if model_size >= 150:
        print(f'  - Size {model_size:.2f}MB is >= 150MB')
    if accuracy < 0.62:
        print(f'  - Accuracy {accuracy:.4f} is < 0.62')
