import struct
import json

def parse_varint(data, offset):
    """Parse a SQLite varint from data at offset, return (value, bytes_read)"""
    value = 0
    shift = 0
    i = 0
    while i < 9:  # varints are at most 9 bytes
        if offset + i >= len(data):
            return None, None
        byte = data[offset + i]
        value |= (byte & 0x7f) << shift
        if byte < 0x80:
            return value, i + 1
        shift += 7
        i += 1
    return value, 9

def parse_record(data, offset, max_len):
    """Parse a SQLite record starting at offset"""
    # First byte(s) is record header size as varint
    header_size, bytes_read = parse_varint(data, offset)
    if header_size is None:
        return None
    
    record_start = offset
    record_data = data[record_start:record_start + max_len]
    
    # Parse header to get column types
    header_end = record_start + header_size
    col_idx = 0
    col_types = []
    pos = record_start + bytes_read
    
    while pos < header_end:
        type_code, bytes_read = parse_varint(data, pos)
        if type_code is None:
            break
        col_types.append(type_code)
        pos += bytes_read
    
    # Now parse the columns
    pos = header_end
    columns = []
    
    for type_code in col_types:
        if type_code == 0:  # NULL
            columns.append(None)
        elif type_code == 1:  # int8
            if pos + 1 <= len(data):
                columns.append(struct.unpack('>b', data[pos:pos+1])[0])
                pos += 1
        elif type_code == 2:  # int16
            if pos + 2 <= len(data):
                columns.append(struct.unpack('>h', data[pos:pos+2])[0])
                pos += 2
        elif type_code == 3:  # int24
            if pos + 3 <= len(data):
                val = int.from_bytes(data[pos:pos+3], 'big', signed=True)
                columns.append(val)
                pos += 3
        elif type_code == 4:  # int32
            if pos + 4 <= len(data):
                columns.append(struct.unpack('>i', data[pos:pos+4])[0])
                pos += 4
        elif type_code == 5:  # int48
            if pos + 6 <= len(data):
                val = int.from_bytes(data[pos:pos+6], 'big', signed=True)
                columns.append(val)
                pos += 6
        elif type_code == 6:  # int64
            if pos + 8 <= len(data):
                columns.append(struct.unpack('>q', data[pos:pos+8])[0])
                pos += 8
        elif type_code == 7:  # IEEE float
            if pos + 8 <= len(data):
                columns.append(struct.unpack('>d', data[pos:pos+8])[0])
                pos += 8
        elif type_code >= 12 and type_code % 2 == 0:  # blob: length = (type_code - 12) / 2
            blob_len = (type_code - 12) // 2
            if pos + blob_len <= len(data):
                columns.append(data[pos:pos+blob_len])
                pos += blob_len
        elif type_code >= 13 and type_code % 2 == 1:  # text: length = (type_code - 13) / 2
            str_len = (type_code - 13) // 2
            if pos + str_len <= len(data):
                try:
                    columns.append(data[pos:pos+str_len].decode('utf-8', errors='ignore'))
                except:
                    columns.append(data[pos:pos+str_len])
                pos += str_len
    
    return columns

import struct

def parse_varint(data, offset):
    """Parse a SQLite varint from data at offset, return (value, bytes_read)"""
    value = 0
    shift = 0
    i = 0
    while i < 9:
        if offset + i >= len(data):
            return None, None
        byte = data[offset + i]
        value |= (byte & 0x7f) << shift
        if byte < 0x80:
            return value, i + 1
        shift += 7
        i += 1
    return value, 9

def parse_record(data, offset, max_len):
    """Parse a SQLite record starting at offset"""
    if max_len <= 0:
        return None
    
    # First byte(s) is record header size as varint
    header_size, bytes_read = parse_varint(data, offset)
    if header_size is None or header_size <= 0:
        return None
    
    # Parse header to get column types
    header_end = offset + header_size
    if header_end > offset + max_len:
        return None
    
    col_types = []
    pos = offset + bytes_read
    
    while pos < header_end and len(col_types) < 10:
        type_code, bytes_read = parse_varint(data, pos)
        if type_code is None:
            break
        col_types.append(type_code)
        pos += bytes_read
    
    # Now parse the columns
    pos = header_end
    columns = []
    
    for type_code in col_types:
        if pos >= offset + max_len:
            break
        
        if type_code == 0:  # NULL
            columns.append(None)
        elif type_code == 1:  # int8
            if pos + 1 <= offset + max_len:
                columns.append(struct.unpack('>b', data[pos:pos+1])[0])
                pos += 1
        elif type_code == 2:  # int16
            if pos + 2 <= offset + max_len:
                columns.append(struct.unpack('>h', data[pos:pos+2])[0])
                pos += 2
        elif type_code == 3:  # int24
            if pos + 3 <= offset + max_len:
                val = int.from_bytes(data[pos:pos+3], 'big', signed=True)
                columns.append(val)
                pos += 3
        elif type_code == 4:  # int32
            if pos + 4 <= offset + max_len:
                columns.append(struct.unpack('>i', data[pos:pos+4])[0])
                pos += 4
        elif type_code == 5:  # int48

                columns.append(val)
