        x_start = col * square_size

import chess

# Detected pieces from image analysis
white_pieces = {'a6', 'a8', 'b7', 'c6', 'c8', 'd5', 'e2', 'e6', 'f5', 'f7', 'g2', 'h1', 'h5'}

board = chess.Board()
board.clear()

for square in white_pieces:
    board.set_piece_at(chess.parse_square(square), chess.Piece(chess.PAWN, chess.WHITE))

for square in black_pieces:
    board.set_piece_at(chess.parse_square(square), chess.Piece(chess.PAWN, chess.BLACK))

board.turn = chess.WHITE

# Analyze all legal moves
promotion_moves = []
check_moves = []
other_moves = []

for move in board.legal_moves:
    board_copy = board.copy()
    board_copy.push(move)
    
    if move.promotion is not None:
        promotion_moves.append(move.uci())
    elif board_copy.is_check():
        check_moves.append(move.uci())
    else:
        other_moves.append(move.uci())

# Promotions are typically winning, so prefer those
best_moves = promotion_moves if promotion_moves else check_moves if check_moves else other_moves

print('Best moves found:')
for move in best_moves:
    print(move)

# Write to file
with open('/app/move.txt', 'w') as f:
    for move in best_moves:
        f.write(move + '\n')

print('\nWritten to /app/move.txt')
