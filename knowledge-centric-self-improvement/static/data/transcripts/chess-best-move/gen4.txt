from PIL import Image
import os

# Load the chess board image
img = Image.open('/app/chess_board.png')
img_array = img.load()

# The image is 640x640, so each square is 640/8 = 80 pixels
square_size = 80

# Define piece identification based on color differences
# We need to detect: dark pieces (black), light pieces (white), and empty squares

def get_square_color(rank, file):
    """Get representative color of a square"""
    x = file * square_size + square_size // 2
    y = rank * square_size + square_size // 2
    return img_array[x, y]

def is_black_piece(color):
    """Check if square contains a black piece"""
    r, g, b = color
    return r < 100 and g < 100 and b < 100

def is_white_piece(color):
    """Check if square contains a white piece"""
    r, g, b = color
    return r > 200 and g > 200 and b > 200

def is_light_square(color):
    """Light square of board"""
    r, g, b = color
    return r > 200 and 200 < g < 230 and b > 150

def is_dark_square(color):
    """Dark square of board"""
    r, g, b = color
    return 150 < r < 200 and 100 < g < 150 and 80 < b < 110

# Analyze board
print("Board analysis (. = empty, B = black, W = white)")
print("Rank 8 (index 0) at top to Rank 1 (index 7) at bottom")
print("File a (index 0) at left to File h (index 7) at right\n")

board = []
for rank in range(8):
    row = []
    line = f"Rank {8-rank}: "
    for file in range(8):
        color = get_square_color(rank, file)
        
        if is_black_piece(color):
            row.append('b')  # black piece
            line += "B "
        elif is_white_piece(color):
            row.append('w')  # white piece
            line += "W "
        else:
            row.append('.')  # empty
            line += ". "
    
    board.append(row)
    print(line)

print("\nFiles:  a b c d e f g h")

# Convert to FEN-like notation for analysis
print("\nDetected positions:")
piece_positions = {}
for rank in range(8):
    for file in range(8):
        if board[rank][file] != '.':
            square_name = chr(ord('a') + file) + str(8 - rank)
            piece_type = "Black" if board[rank][file] == 'b' else "White"
            print(f"  {square_name}: {piece_type}")
