img = Image.open('/app/chess_board.png')

# Detailed analysis using histogram per square
board_analysis = {}

for rank in range(8):
    for file in range(8):
        x_start = file * square_size
        y_start = (7 - rank) * square_size
        x_end = x_start + square_size
        y_end = y_start + square_size
        
        # Collect all RGB values in this square
        rgb_values = []
        for x in range(x_start, x_end, 2):
            for y in range(y_start, y_end, 2):
                r, g, b = pixels[x, y]
                rgb_values.append((r, g, b))
        
        if rgb_values:
            # Calculate statistics
            rs = [v[0] for v in rgb_values]
            gs = [v[1] for v in rgb_values]
            bs = [v[2] for v in rgb_values]
            
            avg_r = sum(rs) / len(rs)
            avg_g = sum(gs) / len(gs)
            avg_b = sum(bs) / len(bs)
            
            std_r = (sum((x - avg_r) ** 2 for x in rs) / len(rs)) ** 0.5
            std_g = (sum((x - avg_g) ** 2 for x in gs) / len(gs)) ** 0.5
            std_b = (sum((x - avg_b) ** 2 for x in bs) / len(bs)) ** 0.5
            
            avg_brightness = (avg_r + avg_g + avg_b) / 3
            complexity = std_r + std_g + std_b
            
            board_analysis[(file, rank)] = {
                'avg': avg_brightness,
                'complexity': complexity,
                'r': avg_r, 'g': avg_g, 'b': avg_b
            }

# Print with piece detection heuristic

print()
