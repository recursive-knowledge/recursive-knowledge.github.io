# Each cell is: payload_length (varint), rowid (varint), then data
# Data format: type bytes (varint) followed by typed values

def parse_varint(data, offset):
    """Parse SQLite varint"""
    result = 0
    shift = 0
    while offset < len(data):
        byte = data[offset]
        offset += 1
        result |= (byte & 0x7f) << shift
        if not (byte & 0x80):
            break
        shift += 7
    return result, offset

def parse_sqlite_record(data, offset, max_len=200):
    """Parse a SQLite record"""
    # First, get header length (varint)
    header_len, offset = parse_varint(data, offset)
    header_start = offset
    
    # Parse type array
    types = []
    type_offset = header_start
    while type_offset < header_start + header_len - 1:
        type_code, type_offset = parse_varint(data, type_offset)
        types.append(type_code)
    
    # Now parse values
    value_offset = header_start + header_len
    values = []
    
    for type_code in types:
        if type_code == 0:  # NULL
            values.append(None)
        elif type_code == 1:  # int8
            values.append(struct.unpack('>b', data[value_offset:value_offset+1])[0])
            value_offset += 1
        elif type_code == 2:  # int16
            values.append(struct.unpack('>h', data[value_offset:value_offset+2])[0])
            value_offset += 2
        elif type_code == 3:  # int24
            val = int.from_bytes(data[value_offset:value_offset+3], 'big', signed=True)
            values.append(val)
            value_offset += 3
        elif type_code == 4:  # int32
            values.append(struct.unpack('>i', data[value_offset:value_offset+4])[0])
            value_offset += 4
        elif type_code == 5:  # int48
            val = int.from_bytes(data[value_offset:value_offset+6], 'big', signed=True)
            values.append(val)
            value_offset += 6
        elif type_code == 6:  # int64
            values.append(struct.unpack('>q', data[value_offset:value_offset+8])[0])
            value_offset += 8
        elif type_code == 7:  # float64
            values.append(struct.unpack('>d', data[value_offset:value_offset+8])[0])
            value_offset += 8
        elif type_code >= 12 and type_code % 2 == 0:  # blob
            length = (type_code - 12) // 2
            values.append(data[value_offset:value_offset+length])
            value_offset += length
        elif type_code >= 13 and type_code % 2 == 1:  # text
            length = (type_code - 13) // 2
            values.append(data[value_offset:value_offset+length].decode('utf-8', errors='ignore'))
            value_offset += length
    
    return values
