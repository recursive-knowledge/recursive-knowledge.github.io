import struct
import json
import re

# Read the database file
with open('/app/trunc.db', 'rb') as f:
    data = f.read()

# Look for patterns: testwordXX followed by a value
# From the hex dump, I can see records like:
# testword09 followed by some bytes
# Let's search for all testwordXX patterns and try to extract values

records = []

# Pattern: testwordXX where XX is digits
pattern = rb'testword(\d+)'
matches = list(re.finditer(pattern, data))

print(f"Found {len(matches)} potential records")

for match in matches:
    start = match.start()
    word_str = match.group(0).decode('latin1')
    # Look at bytes after the word
    end = start + len(match.group(0))
    
    # Read next 20 bytes to see the structure
    context = data[end:end+20]
    print(f"  {word_str} at offset {start}, next bytes: {context.hex()}")
    
    # Try to extract a numeric value
    # Look for patterns: the value might be encoded as varint or direct bytes
    # Let's check if there's a numeric pattern
    
    # Extract the digits from testwordXX
    match_digits = re.search(rb'testword(\d+)', data[start:end+10])
    if match_digits:
        digit_str = match_digits.group(1).decode('latin1')
        value = int(digit_str)
        records.append({"word": word_str, "value": value})
        print(f"    -> Extracted: word={word_str}, value={value}")

print(f"\nTotal records extracted: {len(records)}")
print("Records:")
for r in sorted(records, key=lambda x: int(x['word'].replace('testword', ''))):
    print(f"  {r}")

# Write to JSON
with open('/app/recover.json', 'w') as f:
    json.dump(records, f)
