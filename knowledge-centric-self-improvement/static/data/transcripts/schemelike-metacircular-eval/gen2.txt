Write a file eval.scm that is a metacircular evaluator capable of interpreting the scheme-like language implemented by interp.py.
The eval.scm file should read exactly one line from STDIN, this will be a file path to a new .scm file that should be interpreted.
Remaining input should be redirected to the interpreted-program, and any output from the program should be sent back to STDOUT.
Your interpreter should be able to interpret (1) each of the test programs in test/, and (2) itself.
For example, the following three commands should all do the same thing:
echo '(+ 7 8)' | python3 interp.py test/calculator.scm
echo -e 'test/calculator.scm\n(+ 7 8)' | python3 interp.py eval.scm
echo -e 'eval.scm\ntest/calculator.scm\n(+ 7 8)' | python3 interp.py eval.scm
The first of these will add 7 and 8 because that is what calculator.scm does.
The second will run eval.scm, which takes test/calculator.scm as the file to run, and runs it; that program then computes 7+8.
The third will run eval.scm, which then takes eval.scm as the input, the outer eval.scm interprets the inner eval.scm which then runs the calculator as above.

class Environment:

        self.parent = parent

        self.bindings[name] = value

def parse_expr(tokens, index):
    if index >= len(tokens):
        raise SchemeError("Unexpected end of input")

    token = tokens[index]

    if token == "(":
        index += 1
        elements = []
        while index < len(tokens) and tokens[index] != ")":
            expr, index = parse_expr(tokens, index)
            elements.append(expr)
        if index >= len(tokens):
            raise SchemeError("Missing closing parenthesis")
        index += 1

        result = None
        for i in range(len(elements) - 1, -1, -1):
            result = Pair(elements[i], result)
        return result, index

    elif token == ")":
        raise SchemeError("Unexpected closing parenthesis")

    elif isinstance(token, String):
        return token, index + 1

    elif token == "'":
        expr, index = parse_expr(tokens, index + 1)
        return Pair("quote", Pair(expr, None)), index

    else:
        # Try to parse as number
        try:
            return int(token), index + 1
        except ValueError:
            pass

        # Check for boolean literals
        if token == "#t":
            return True, index + 1
        elif token == "#f":
            return False, index + 1

        # Otherwise it's a symbol
        return token, index + 1


def parse(tokens):
    expr, _ = parse_expr(tokens, 0)
    return expr

(define (map-fn fn lst)
  (if (null? lst) '()
      (cons (fn (car lst)) (map-fn fn (cdr lst)))))

;;; Core evaluator
(define (eval-expr expr env)
  (cond
    ((number? expr) expr)
    ((eq? expr #t) #t)
    ((eq? expr #f) #f)
    ((string? expr) expr)
    ((symbol? expr)
     (let ((val (lookup expr env)))
       (if val val expr)))
    ((and (pair? expr) (eq? (car expr) 'quote))
     (car (cdr expr)))
    ((and (pair? expr) (eq? (car expr) 'define))
     (define-variable (car (cdr expr)) (eval-expr (car (cdr (cdr expr))) env)) #f)
    ((and (pair? expr) (eq? (car expr) 'if))
     (if (eval-expr (car (cdr expr)) env)
         (eval-expr (car (cdr (cdr expr))) env)
         (if (pair? (cdr (cdr (cdr expr))))
             (eval-expr (car (cdr (cdr (cdr expr)))) env)
             #f)))
    ((and (pair? expr) (eq? (car expr) 'lambda))
     (cons 'lambda (cons (car (cdr expr)) (cons (cdr (cdr expr)) (cons env '())))))
    ((and (pair? expr) (or (eq? (car expr) 'begin) (eq? (car expr) 'progn)))
     (eval-sequence (cdr expr) env))
    ((and (pair? expr) (eq? (car expr) 'cond))
     (eval-cond-clauses (cdr expr) env))
    ((and (pair? expr) (eq? (car expr) 'let))
     (eval-let-expr (car (cdr expr)) (cdr (cdr expr)) env))
    ((pair? expr)
     (let ((func (eval-expr (car expr) env)) (args (cdr expr)))
       (apply-fn func args env)))
    (else expr)))

(define (eval-sequence exprs env)
  (if (null? exprs) #f
      (if (null? (cdr exprs))
          (eval-expr (car exprs) env)
          (progn (eval-expr (car exprs) env) (eval-sequence (cdr exprs) env)))))

(define (eval-cond-clauses clauses env)
  (if (null? clauses) #f
      (let ((clause (car clauses)))
        (if (eval-expr (car clause) env)
            (eval-sequence (cdr clause) env)
            (eval-cond-clauses (cdr clauses) env)))))

(define (eval-let-expr bindings body env)
  (if (null? bindings) (eval-sequence body env)
      (let ((binding (car bindings)))
        (eval-let-expr (cdr bindings) body
                       (extend-env (cons (car binding) '())
                                   (cons (eval-expr (car (cdr binding)) env) '())
                                   env)))))

;;; Function application
(define (apply-fn func args env)
  (cond
    ((and (pair? func) (eq? (car func) 'lambda))
     (let ((params (car (cdr func)))
           (body (car (cdr (cdr func))))
           (cenv (car (cdr (cdr (cdr func))))))
       (let ((eargs (map-fn (lambda (a) (eval-expr a env)) args)))
         (eval-sequence body (extend-env params eargs cenv)))))
    ((symbol? func) (apply-builtin func args env))
    (else #f)))
