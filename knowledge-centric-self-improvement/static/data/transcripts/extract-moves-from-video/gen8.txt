            capture_output=True,
            text=True,
            timeout=3
        )
        text = result.stdout.strip()
        if not text:

        # Find all lines that contain > followed by a command
        # The pattern is: >COMMAND
        for line in text.split('\n'):
            # Look for > character (might be OCR'd as similar char)
            if '>' in line:
                # Extract what comes after >
                parts = line.split('>')
                for j in range(1, len(parts)):
                    potential_cmd = parts[j].strip()
                    
                    # Extract first word or short phrase (until newline or next >)
                    # Take only the first meaningful word/command
                    cmd_match = re.match(r'^([a-zA-Z\s]+?)(?:\s{2,}|$|>)', potential_cmd)
