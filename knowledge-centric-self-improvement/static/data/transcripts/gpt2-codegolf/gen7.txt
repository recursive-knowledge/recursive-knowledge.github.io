#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define VOCAB_SIZE 50257
#define EMBED_DIM 768
#define N_LAYER 12
#define N_HEAD 12
#define FFN_DIM 3072

typedef struct {
    float *data;
    int size;
} Tensor;

Tensor* read_weights(const char *ckpt_path) {
    FILE *f = fopen(ckpt_path, "rb");
    if (!f) return NULL;
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    fseek(f, 0, SEEK_SET);
    Tensor *t = malloc(sizeof(Tensor));
    t->data = malloc(size);
    t->size = size / 4;
    fread(t->data, 1, size, f);
    fclose(f);
    return t;
}

int* read_vocab(const char *vocab_path) {
    int *vocab = malloc(VOCAB_SIZE * sizeof(int));
    memset(vocab, 0, VOCAB_SIZE * sizeof(int));
    FILE *f = fopen(vocab_path, "r");
    if (!f) return vocab;
    char line[256];
    int idx = 0;
    while (fgets(line, 256, f) && idx < VOCAB_SIZE) {
        if (line[0] == '#') continue;
        idx++;
    }
    fclose(f);
    return vocab;
}

void softmax(float *logits, int size) {
    float max_val = logits[0];
    for (int i = 1; i < size; i++) {
        if (logits[i] > max_val) max_val = logits[i];
    }
    float sum = 0;
    for (int i = 0; i < size; i++) {
        logits[i] = exp(logits[i] - max_val);
        sum += logits[i];
    }
    for (int i = 0; i < size; i++) {
        logits[i] /= sum;
    }
}

int argmax(float *logits, int size) {

void softmax(float *logits, int size) {
    float max_val = logits[0];
    for (int i = 1; i < size; i++) {
        if (logits[i] > max_val) max_val = logits[i];

    float sum = 0;
