  FILE*f=fopen(argv[2],"rb");
  if(!f)return 1;
  char line[256];
  while(fgets(line,255,f)&&vocab_sz<50001){
    line[strcspn(line,"\n")]=0;
    if(line[0]=='#')continue;
    vocab[vocab_sz].k=malloc(strlen(line)+1);
    strcpy(vocab[vocab_sz].k,line);
    vocab[vocab_sz].v=vocab_sz++;
