# Create a script to boot Alpine with proper boot command
$ cd /app && cat > boot_alpine.sh << 'EOF'
#!/bin/bash
# Kill any existing QEMU processes
for pid in $(find /proc -maxdepth 2 -name cmdline 2>/dev/null -exec grep -l qemu {} \; 2>/dev/null | sed 's|/proc/\([0-9]*\)/.*|\1|' | sort -u); do
  kill -9 $pid 2>/dev/null
done
sleep 2

# Start QEMU with serial input
rm -f qemu.log
nohup qemu-system-x86_64 -m 1024 -cdrom alpine.iso -drive file=alpine-disk.qcow2,format=qcow2 -net nic -net user,hostfwd=tcp::2222-:22 -nographic > qemu.log 2>&1 &
QEMU_PID=$!
sleep 3

# Send boot command to QEMU via monitor
echo 'boot' | nc localhost 2222 2>/dev/null || true
sleep 2
EOF
chmod +x boot_alpine.sh
echo 'Script created'
